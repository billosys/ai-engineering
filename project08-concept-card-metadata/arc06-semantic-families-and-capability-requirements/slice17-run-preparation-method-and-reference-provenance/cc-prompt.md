def leaf_state($x):

  elif ($x|type)=="object" and ($x|length)==0 then "empty-mapping"
  elif ($x|type)=="array" then "populated-list"
  elif ($x|type)=="object" then "populated-mapping"

def field_state($v; $path):
  ($path|split(".")) as $p |
  if ($p|length)==1 then
    if ($v|has($p[0])|not) then "absent" else leaf_state($v[$p[0]]) end
  elif ($v|has($p[0])|not) then "parent-absent"
