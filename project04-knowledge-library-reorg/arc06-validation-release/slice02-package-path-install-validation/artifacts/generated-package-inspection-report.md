```sh
for z in collaboration-framework.zip rust-guidelines.zip go-guidelines.zip \
  cpp-guidelines.zip javascript-deno-guidelines.zip erlang-guidelines.zip \
  cobalt-guidelines.zip visual-design-system.zip tailwindcss.zip \
  deno-js-linter.zip biome-js-linter.zip biome-linter.zip; do
  printf '%s\t' "$z"
  unzip -Z1 "$z" | awk -F/ 'NR==1 {root=$1} END {print root, NR " entries"}'
done
```

Entrypoint inspection:

```sh
for z in collaboration-framework.zip rust-guidelines.zip go-guidelines.zip \
  cpp-guidelines.zip javascript-deno-guidelines.zip erlang-guidelines.zip \
  cobalt-guidelines.zip visual-design-system.zip tailwindcss.zip \
  deno-js-linter.zip biome-js-linter.zip biome-linter.zip; do
  printf '== %s ==\n' "$z"
  unzip -Z1 "$z" | rg '(^[^/]+/SKILL[^/]*\.md$|^collaboration-framework/SKILL.md$)'
done
```
